# Floors AND ceilings, on purpose.
#
# These were all bare `>=` once, which quietly means "whatever is newest on
# the day someone installs". That is not a stable base for a release: at the
# time of writing a fresh install pulled opencv-python 5.x and psutil 7.x,
# two major versions nobody had ever run the app against.
#
# The upper bounds pin the MAJOR only. Patches and minors still flow in, so
# real fixes are not blocked; a new major is a deliberate decision with a
# real video tested afterwards, not something a new contributor discovers by
# accident.
#
# Raising one of these ceilings is a normal thing to do. Do it on purpose,
# and process an actual video before merging it.

# Extractors break whenever a site changes, so this is the one dependency
# you always want newest. No ceiling.
yt-dlp>=2026.8.19

# ---- transcription and vision --------------------------------------------
# MFCC features for the active-speaker model. Tiny and pure Python; the
# model was trained against this library's defaults, so it is not
# interchangeable with another MFCC implementation.
python_speech_features>=0.6,<1
faster-whisper>=1.2.1,<2
# faster-whisper's engine (CTranslate2) is a CUDA 12 build and needs cuBLAS 12
# on an NVIDIA GPU; PyTorch's CUDA 13 build only has cuBLAS 13. Without this,
# Whisper falls back to the CPU (issue #111). Windows only: that's where the
# app ships, and the Linux image runs on the CPU.
nvidia-cublas-cu12>=12.4,<13 ; sys_platform == "win32"
ultralytics>=8.4.116,<9
# <5 matters. opencv-python 5.0 is out and pip takes it without this.
opencv-python>=4.14.0.94,<5
# On-screen text in a gaming stream (analysis/game_text.py): PaddleOCR's
# models on the onnxruntime Piper already brings. 1.4 carries its models in
# the wheel; the later "rapidocr" package downloads them at first use, which
# an installed copy must never do.
rapidocr-onnxruntime>=1.4.4,<1.5
# Runs the two voice models behind the second speaker's caption colour
# (analysis/voice_turns.py). Already installed as a dependency of
# faster-whisper and Piper; pinned now that it is used directly.
onnxruntime>=1.26,<2

# ---- AI dubbing --------------------------------------------------------

# copy simply reported dubbing unavailable, because the code shelled out to

# in-process, so it can ship.
#
# Voices are NOT bundled: ~60 MB each across 15 languages, downloaded on
# demand into data/voices/ the first time a language is dubbed.
